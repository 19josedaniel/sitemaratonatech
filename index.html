    function analisarMensagem(texto) {
    const termosUrgentes = ["urgente", "bloqueio", "bloqueada", "pix", "senha", "atualize"];
    let pontuacao = 0;

    termosUrgentes.forEach(termo => {
        if (texto.toLowerCase().includes(termo)) pontuacao += 20;
    });

    if (texto.includes("http://") || (!texto.includes("https://") && texto.includes(".com"))) {
        pontuacao += 40;
    }

    return { risco: pontuacao > 40 ? "ALTO" : "BAIXO", score: Math.min(pontuacao, 100) };
}
